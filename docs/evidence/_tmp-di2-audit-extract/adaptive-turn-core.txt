  603|type ClarificationBlock = "budget_exhausted" | "depth_exhausted";
  604|
  605|function clarificationBlock(
  606|  session: POJUSessionState,
  607|  plan: AdaptiveQuestionPlan,
  608|  answerResolution: boolean,
  609|): ClarificationBlock | null {
  610|  const u = session.decision_understanding;
  611|  const bp = session.decision_blueprint;
  612|  if (!u || !bp || plan.budget_cost === 0) return null;
  613|  if (
  614|    u.budget.remaining_hard_budget < plan.budget_cost ||
  615|    u.budget.session_asked_count >= u.budget.hard_budget_limit
  616|  ) {
  617|    return "budget_exhausted";
  618|  }
  619|  // Binding what the user already said is not probing deeper into a direction.
  620|  if (answerResolution) return null;
  621|  const direction = bp.directions.find((d) => d.direction_id === plan.direction_id);
  622|  const cap = direction?.depth_soft_cap ?? bp.question_budget.depth_per_direction;
  623|  if (plan.dir_ids.some((dirId) => (u.budget.depth_by_dir[dirId] ?? 0) >= cap)) {
  624|    return "depth_exhausted";
  625|  }
  626|  return null;

  629|/**
  630| * Clarifications are questions: charged and bound by the hard budget; aspect /
  631| * insight follow-ups are also bound by the per-direction depth. A clarification
  632| * that would be newly sent past its limit is not asked: its item closes auditably
  633| * (never written as fact) and the next clarification or the planner decides,
  634| * which reaches readiness once nothing askable is left. Re-presenting an
  635| * already-sent question and free goal_change confirmations are never blocked.
  636| */
  637|async function closeBlockedClarifications(
  638|  result: AdaptiveTurnResult,
  639|  input: AdaptiveTurnInput,
  640|): Promise<AdaptiveTurnResult> {
  641|  let r = result;
  642|  for (let guard = 0; guard < 64; guard++) {
  643|    const plan = r.plan;
  644|    const u = r.session.decision_understanding;
  645|    if (!plan || !u || !r.session.decision_blueprint) return r;
  646|    const isPending = isPendingClarificationPlan(plan);
  647|    if (!isPending && !isIntakeControlPlan(plan)) return r;
  648|    if (u.last_sent_question?.plan_id === plan.plan_id) return r;
  649|    const block = clarificationBlock(r.session, plan, isPending);
  650|    if (!block) return r;
