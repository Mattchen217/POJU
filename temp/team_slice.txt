            function escapeHtml(value) {
              return String(value)
                .replace(/&/g, '&amp;')
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;')
                .replace(/"/g, '&quot;')
                .replace(/'/g, '&#39;');
            }

            function renderTeamCard(member, avatarSrc, index) {

                index +
                '">' +
                '<div class="v2-team-avatar-wrap" aria-hidden="true">' +
                '<img class="v2-team-avatar" src="' +
                avatarSrc +
                '" alt="" width="176" height="176" loading="lazy" decoding="async"/>' +
                '</div>' +
                '<div class="absolute top-0 right-0 p-6 opacity-10 pointer-events-none">' +
                '<span class="material-icons-outlined text-6xl text-primary">format_quote</span>' +
